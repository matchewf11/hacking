-------------------------------------------------------------
1. BASIC GET REQUEST
   Fetch a resource, pretty-print the response, filter a
   JSON field, and chain with query args.
-------------------------------------------------------------

# Simple fetch with pretty-printing
wurl get "https://api.example.com/users" --pretty-print

# Scoped path (avoids repeating the base URL)
wurl get "https://api.example.com" --path users/42

# Query string args
wurl get "https://api.example.com/search" --args q=alice,page=1,limit=20
wurl get "https://api.example.com/search" -a q=alice,page=1,limit=20


-------------------------------------------------------------
2. POST / PUT WITH A JSON BODY
   := auto-converts values (booleans, numbers).
   =  forces string.
   Dot notation builds nested objects.
-------------------------------------------------------------

# Create a user
wurl post "https://api.example.com/users" \
    --json name="Alice" \
    --json email="alice@example.com" \
    --json role="admin" \
    --json active:=true \
    --json age:=30

# Nested object (foo.bar builds { "foo": { "bar": ... } })
wurl post "https://api.example.com/profiles" \
    --json user.name="Alice" \
    --json user.verified:=true \
    --json settings.theme="dark"

# Update (PUT)
wurl put "https://api.example.com/users/42" \
    --json role="member" \
    --json active:=false


-------------------------------------------------------------
3. AUTHENTICATION FLOW
   Test login happy-path: status, JWT shape, secure cookie,
   and response time. Then test the failure cases.
-------------------------------------------------------------

include "env.wurl"
replace API_BASE
replace TEST_PASSWORD

group "auth"
    test "login returns a JWT and secure session cookie"
        post "${API_BASE}/auth/login"
            body.email = "alice@example.com"
            body.password = ${TEST_PASSWORD}
        assert status 200
        assert body.token present
        assert body.token matches r"^[A-Za-z0-9\-_]+\.[A-Za-z0-9\-_]+\.[A-Za-z0-9\-_]+$"
        assert header.content-type contains "json"
        assert cookie.session present
        assert cookie.session.httponly
        assert cookie.session.secure
        assert cookie.session.samesite equals "Strict"
        assert duration lt 800
    end

    test "rejects wrong password"
        post "${API_BASE}/auth/login"
            body.email = "alice@example.com"
            body.password = "wrong"
        assert status 401
        assert body.error equals "invalid_credentials"
        assert not body.token
        assert not cookie.session present
    end

    test "rejects missing fields"
        post "${API_BASE}/auth/login"
            body.email = "alice@example.com"
        assert status 400
        assert body.errors present
        assert body.errors length gt 0
    end

    test "logout clears session cookie"
        post "${API_BASE}/auth/logout"
        assert status 204
        assert body empty
        assert cookie.session absent
    end
end


-------------------------------------------------------------
4. CRUD RESOURCE LIFECYCLE
   Run create → read → update → delete sequentially inside
   one group. Tests in the same group are ordered; tests
   across groups run in parallel.
-------------------------------------------------------------

include "env.wurl"
replace API_BASE
replace ADMIN_TOKEN

group "posts CRUD"
    test "create a post"
        post "${API_BASE}/posts"
            body.title = "Hello Wurl"
            body.body  = "First post via wurl test suite."
            body.published:=false
        assert status 201
        assert body.id present
        assert body.title equals "Hello Wurl"
        assert body.published false
        assert header.location present
    end

    test "read the post back"
        get "${API_BASE}/posts/1"
        assert status 200
        assert body.title equals "Hello Wurl"
        assert body.author present
        assert not body.deleted
    end

    test "update the post"
        put "${API_BASE}/posts/1"
            body.published:=true
        assert status 200
        assert body.published true
        assert body.updated_at present
    end

    test "delete the post"
        delete "${API_BASE}/posts/1"
        assert status 204
        assert body empty
    end

    test "deleted post returns 404"
        get "${API_BASE}/posts/1"
        assert status 404
        assert body.error present
    end
end


-------------------------------------------------------------
5. PAGINATION & ARRAY ASSERTIONS
   Validate list endpoints: shape of the envelope, first
   item fields, length constraints, and link headers.
-------------------------------------------------------------

include "env.wurl"
replace API_BASE

group "user list pagination"
    test "first page returns expected shape"
        get "${API_BASE}/users" -a page=1,per_page=10
        assert status 200
        assert body.data present
        assert body.data length 10
        assert body.data[0].id present
        assert body.data[0].email present
        assert body.meta.total gt 0
        assert body.meta.page equals 1
        assert header.link present
        assert header.link contains "rel=\"next\""
    end

    test "last page has no next link"
        get "${API_BASE}/users" -a page=999,per_page=10
        assert status 200
        assert body.data length lte 10
        assert not header.link contains "rel=\"next\""
    end

    test "empty filter returns empty array"
        get "${API_BASE}/users" -a role=nonexistent
        assert status 200
        assert body.data empty
        assert body.meta.total equals 0
    end
end


-------------------------------------------------------------
6. ENVIRONMENT-DRIVEN SETUP & REGEX VALIDATION
   Pull secrets from env, validate complex field formats,
   and assert on nested structures and response timing.
-------------------------------------------------------------

include "common.wurl"
replace BASE_URL          # pulled from env if no value given
replace EXPECTED_ORG
replace 5 MAX_LATENCY_MS  # override with literal value

group "profile validation"
    test "profile fields pass format checks"
        get "${BASE_URL}/me"
        assert status 200

        # string equality via env var
        assert body.org equals ${EXPECTED_ORG}

        # email format
        assert body.email matches r"^[\w.+\-]+@[\w.\-]+\.[a-z]{2,}$"

        # UUID v4
        assert body.id matches r"^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$"

        # sensitive fields must be absent
        assert not body.password
        assert not body.password_hash

        # timestamps present
        assert body.created_at present
        assert body.updated_at present

        # nested boolean
        assert body.settings.notifications.email true

        # performance budget
        assert duration lte ${MAX_LATENCY_MS}
    end

    test "content-type is always JSON"
        get "${BASE_URL}/me"
        assert header.content-type matches r"application/(json|problem\+json)"
        assert not header.x-deprecated present
    end
end
