-------------------------------------------------------------
1. BASIC GET REQUEST
   Flags: --path, --query/-q, --headers, --cookies/-c
-------------------------------------------------------------

# Simple GET
wurl get "https://api.example.com/users"

# Scoped path
wurl get "https://api.example.com" --path /users/42

# Query args — repeat --query (or -q) for each param
wurl get "https://api.example.com/search" --query q=alice --query page=1 --query limit=20
wurl get "https://api.example.com/search" -q q=alice -q page=1

# Custom request headers
wurl get "https://api.example.com/data" \
    --headers "Authorization=Bearer token123" \
    --headers "Accept=application/json"

# Send cookies
wurl get "https://api.example.com/dashboard" --cookies "session=abc123" -c "theme=dark"


-------------------------------------------------------------
2. POST WITH A JSON BODY
   = auto-converts: integers, booleans (true/false), null,
     and arrays [item1, item2, ...].
   Dot notation builds nested objects.
   Use --json / -j for each field.
-------------------------------------------------------------

# Create a user (values are auto-typed)
wurl post "https://api.example.com/users" \
    --json name=Alice \
    --json email=alice@example.com \
    --json role=admin \
    --json active=true \
    --json age=30

# Nested object — dot notation → { "user": { "name": ... } }
wurl post "https://api.example.com/profiles" \
    --json user.name=Alice \
    --json user.verified=true \
    --json settings.theme=dark

# Array value
wurl post "https://api.example.com/batch" \
    --json ids=[1,2,3] \
    --json flags=[true,false,null]

# Short flag -j
wurl post "https://api.example.com/items" -j name=Widget -j count=5

# POST with path, query, and headers
wurl post "https://api.example.com" \
    --path /orders \
    --json product=Widget \
    --json qty=2 \
    --headers "Authorization=Bearer token123" \
    -q dry_run=true


-------------------------------------------------------------
3. TEST SUITE — STRUCTURE
   From stdin:   cat suite.wurl | wurl test -
   Heredoc:      wurl test - <<'EOF' ... EOF

   Groups contain tests; tests contain a request value and
   any number of assert lines.
-------------------------------------------------------------

group "auth"
    test "login returns 200"
        post "https://api.example.com/auth/login"
        assert status 200
        assert body.token present
        assert duration lt 800
    end

    test "bad password returns 401"
        post "https://api.example.com/auth/login"
        assert status 401
        assert body.error equals "invalid_credentials"
        assert not body.token present
    end

    test "logout clears session"
        post "https://api.example.com/auth/logout"
        assert status 204
        assert body empty
        assert cookie.session absent
    end
end


-------------------------------------------------------------
4. ASSERT TARGETS
   status | duration | body | body.<path> | body[<idx>]
   header.<name> | cookie.<name> | cookie.<name>.<attr>
-------------------------------------------------------------

# Status code
assert status 200
assert not status 404

# Raw response body
assert body empty
assert body contains "success"
assert body equals "ok"

# JSON field (dot-notation path)
assert body.user.name equals "Alice"
assert body.active true
assert body.count gt 0
assert body.score lte 100

# Array indexing
assert body.items[0].id present
assert body.data[0].email present
assert body[0].name equals "Alice"

# Headers (name is lowercased automatically)
assert header.content-type contains "json"
assert header.x-request-id present
assert not header.x-deprecated present

# Cookies — name only, or name + attribute
assert cookie.session present
assert cookie.session equals "abc123"
assert cookie.session.httponly
assert cookie.session.secure
assert cookie.session.samesite equals "Strict"
assert cookie.tracking absent

# Response time in milliseconds
assert duration lt 500
assert duration lte 1000


-------------------------------------------------------------
5. ALL MATCHERS
-------------------------------------------------------------

# Unary
assert body.field present           # field exists / is truthy
assert body.field absent            # field is missing
assert body empty                   # body (or array/string) is empty

# Equality — explicit keyword, shorthand, or implicit (bare value)
assert body.role equals "admin"
assert body.role eq "admin"
assert body.role "admin"            # implicit equals
assert body.count not-equals 0
assert body.count neq 0

# Numeric comparisons
assert body.age gt 18
assert body.age gte 18
assert body.age lt 100
assert body.age lte 99
assert duration lt 500

# String matchers
assert body.name contains "ali"
assert body.name not-contains "bot"
assert body.name starts-with "Al"
assert body.name ends-with "ce"
assert header.content-type contains "application/json"

# Array / string length
assert body.items length 5          # exactly 5
assert body.tags length gt 0        # at least 1
assert body.tags length gte 2
assert body.tags length lt 10
assert body.tags length lte 20

# Regex — r"..." literal
assert body.email matches r"^[\w.+\-]+@[\w.\-]+\.[a-z]{2,}$"
assert body.id    matches r"^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$"
assert header.content-type matches r"application/(json|problem\+json)"
assert body.token matches r"^[A-Za-z0-9\-_]+\.[A-Za-z0-9\-_]+\.[A-Za-z0-9\-_]+$"


-------------------------------------------------------------
6. NEGATION — assert not <target> [matcher]
-------------------------------------------------------------

assert not status 404

assert not body.error
assert not header.x-deprecated present
assert not cookie.tracking present
assert not body.user.name equals "root"
assert not body.name contains "bot"

7. ASSERT VALUES
   Quoted string:  "hello world"
   Number:         200  3.14  0
   Boolean:        true  false
   Null:           null
   Regex:          r"pattern"
   Bare string:    admin  (unquoted single-word ident)

assert body.role "admin"            # bare string, implicit equals
assert body.role equals admin       # same — bare ident parsed as string
assert body.active true             # boolean
assert body.count equals 0          # number
assert body.field equals null       # null

8. FULL EXAMPLE: CRUD + PAGINATION
-------------------------------------------------------------

group "posts CRUD"
    test "create a post"
        post "https://api.example.com/posts"
        assert status 201
        assert body.id present
        assert body.title present
        assert header.location present
    end

    test "read the post back"
        get "https://api.example.com/posts/1"
        assert status 200
        assert body.title equals "Hello Wurl"
        assert not body.deleted
    end

    test "deleted post returns 404"
        get "https://api.example.com/posts/99999"
        assert status 404
        assert body.error present
    end
end

group "user list"
    test "first page has expected shape"
        get "https://api.example.com/users"
        assert status 200
        assert body.data present
        assert body.data length 10
        assert body.data[0].id present
        assert body.data[0].email present
        assert body.meta.total gt 0
        assert header.link present
    end

    test "empty filter returns empty array"
        get "https://api.example.com/users"
        assert status 200
        assert body.data empty
        assert body.meta.total equals 0
    end
end
